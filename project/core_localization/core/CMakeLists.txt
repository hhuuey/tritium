cmake_minimum_required(VERSION 3.16)
project(fast_livo_core VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FAST_LIVO_FP_MODEL "legacy" CACHE STRING "Floating-point execution model: legacy or strict")
set_property(CACHE FAST_LIVO_FP_MODEL PROPERTY STRINGS legacy strict)
if(NOT FAST_LIVO_FP_MODEL STREQUAL "legacy" AND NOT FAST_LIVO_FP_MODEL STREQUAL "strict")
  message(FATAL_ERROR "Unsupported FAST_LIVO_FP_MODEL")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenCV REQUIRED)
find_package(Sophus REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Boost REQUIRED COMPONENTS thread timer)
find_package(GTSAM REQUIRED)
find_package(vikit_common REQUIRED CONFIG)
find_package(yaml-cpp REQUIRED)
list(APPEND CMAKE_MODULE_PATH /usr/share/cmake/geographiclib /usr/share/cmake/GeographicLib)
find_package(GeographicLib REQUIRED)

file(GLOB BACKEND_SOURCES CONFIGURE_DEPENDS src/backend*.cpp)
add_library(fast_livo_core
  src/config.cpp src/camera_factory.cpp src/engine.cpp src/timestamp.cpp src/validation.cpp src/trace.cpp src/numerics.cpp
  src/IMU_Processing.cpp src/LIVMapper.cpp src/preprocess.cpp src/voxel_map.cpp
  src/vio.cpp src/vio_patch.cpp src/frame.cpp src/visual_point.cpp ${BACKEND_SOURCES})
add_library(fast_livo::core ALIAS fast_livo_core)
set_target_properties(fast_livo_core PROPERTIES EXPORT_NAME core POSITION_INDEPENDENT_CODE ON)
target_compile_features(fast_livo_core PUBLIC cxx_std_17)
target_compile_options(fast_livo_core PRIVATE -O3 -pthread -fexceptions)
target_compile_definitions(fast_livo_core PRIVATE FAST_LIVO_FP_MODEL="${FAST_LIVO_FP_MODEL}")
set_source_files_properties(src/config.cpp src/engine.cpp src/timestamp.cpp src/validation.cpp src/vio_patch.cpp
  PROPERTIES COMPILE_OPTIONS -fno-fast-math)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|aarch64|ARM|AARCH64)")
  target_compile_definitions(fast_livo_core PRIVATE ARM_ARCH)
  target_compile_options(fast_livo_core PRIVATE -march=armv8-a)
  if(FAST_LIVO_FP_MODEL STREQUAL "legacy")
    target_compile_options(fast_livo_core PRIVATE -ffast-math)
  endif()
else()
  target_compile_definitions(fast_livo_core PRIVATE X86_ARCH)
  target_compile_options(fast_livo_core PRIVATE -funroll-loops)
endif()
if(FAST_LIVO_FP_MODEL STREQUAL "strict")
  target_compile_options(fast_livo_core PRIVATE -fno-fast-math -ffp-contract=off)
endif()
include(ProcessorCount)
ProcessorCount(CORE_COUNT)
if(CORE_COUNT GREATER 1)
  if(CORE_COUNT GREATER 4)
    set(CORE_COUNT 4)
  endif()
  target_compile_definitions(fast_livo_core PRIVATE MP_EN MP_PROC_NUM=${CORE_COUNT})
else()
  target_compile_definitions(fast_livo_core PRIVATE MP_PROC_NUM=1)
endif()
target_include_directories(fast_livo_core
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>
    ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS}
  PRIVATE private ${Sophus_INCLUDE_DIRS} ${GeographicLib_INCLUDE_DIRS}
    ${vikit_common_INCLUDE_DIRS} ${vikit_common_INCLUDE_DIR})
target_link_libraries(fast_livo_core PUBLIC Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS}
  PRIVATE OpenMP::OpenMP_CXX ${Sophus_LIBRARIES} ${GeographicLib_LIBRARIES}
    ${vikit_common_LIBRARIES} gtsam yaml-cpp Boost::thread Boost::timer)

install(TARGETS fast_livo_core EXPORT fast_livo_coreTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
add_executable(fast_livo_numerics tools/numerics_info.cpp)
target_link_libraries(fast_livo_numerics PRIVATE fast_livo::core)
install(TARGETS fast_livo_numerics RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT fast_livo_coreTargets NAMESPACE fast_livo:: DESTINATION lib/cmake/fast_livo_core)
configure_package_config_file(cmake/fast_livo_coreConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/fast_livo_coreConfig.cmake INSTALL_DESTINATION lib/cmake/fast_livo_core)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/fast_livo_coreConfigVersion.cmake
  COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fast_livo_coreConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/fast_livo_coreConfigVersion.cmake DESTINATION lib/cmake/fast_livo_core)
