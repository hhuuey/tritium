cmake_minimum_required(VERSION 3.16)
project(fast_livo_ros2 LANGUAGES C CXX)

find_package(ament_cmake REQUIRED)
find_package(fast_livo_core CONFIG REQUIRED)
set(adapter_dependencies
  rclcpp builtin_interfaces std_msgs sensor_msgs geometry_msgs nav_msgs visualization_msgs std_srvs tf2_ros
  cv_bridge pcl_conversions fast_livo_ros2_interfaces)
foreach(dependency IN LISTS adapter_dependencies)
  find_package(${dependency} REQUIRED)
endforeach()

add_executable(fastlivo_mapping src/main.cpp src/config.cpp src/inputs.cpp src/outputs.cpp)
target_compile_features(fastlivo_mapping PRIVATE cxx_std_17)
target_include_directories(fastlivo_mapping PRIVATE include)
target_link_libraries(fastlivo_mapping fast_livo::core)
ament_target_dependencies(fastlivo_mapping ${adapter_dependencies})
install(TARGETS fastlivo_mapping DESTINATION lib/${PROJECT_NAME})
find_package(fast_livo_io CONFIG REQUIRED)
find_package(rosgraph_msgs REQUIRED)
add_executable(fast_livo_replayer src/replay.cpp src/replay_publishers.cpp)
target_compile_features(fast_livo_replayer PRIVATE cxx_std_17)
target_include_directories(fast_livo_replayer PRIVATE include)
target_link_libraries(fast_livo_replayer fast_livo::fast_livo_io)
ament_target_dependencies(fast_livo_replayer
  rclcpp sensor_msgs std_msgs builtin_interfaces rosgraph_msgs cv_bridge pcl_conversions fast_livo_ros2_interfaces)
install(TARGETS fast_livo_replayer DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(yaml-cpp REQUIRED)
  add_executable(fast_livo_receipt_gate validation/receipt_gate.cpp)
  target_compile_features(fast_livo_receipt_gate PRIVATE cxx_std_17)
  target_link_libraries(fast_livo_receipt_gate yaml-cpp)
  ament_target_dependencies(fast_livo_receipt_gate rclcpp nav_msgs std_srvs)
  install(TARGETS fast_livo_receipt_gate DESTINATION lib/${PROJECT_NAME})
  add_executable(fast_livo_compare validation/compare_results.cpp)
  target_compile_features(fast_livo_compare PRIVATE cxx_std_17)
  install(TARGETS fast_livo_compare DESTINATION lib/${PROJECT_NAME})
  ament_add_gtest(imu_conversion_test test/imu_conversion_test.cpp)
  target_compile_features(imu_conversion_test PRIVATE cxx_std_17)
  target_include_directories(imu_conversion_test PRIVATE include)
  target_link_libraries(imu_conversion_test fast_livo::core)
  ament_target_dependencies(imu_conversion_test sensor_msgs)
  ament_add_gtest(message_conversion_test test/message_conversion_test.cpp)
  target_compile_features(message_conversion_test PRIVATE cxx_std_17)
  target_include_directories(message_conversion_test PRIVATE include)
  target_link_libraries(message_conversion_test fast_livo::core yaml-cpp)
  ament_target_dependencies(message_conversion_test ${adapter_dependencies})
endif()

ament_package()
