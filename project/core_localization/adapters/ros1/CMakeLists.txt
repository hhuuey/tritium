cmake_minimum_required(VERSION 3.16)
if(PROJECT_NAME STREQUAL "fast_livo")
  set(FAST_LIVO_ROS1_ROOT_LAYOUT TRUE)
else()
  project(fast_livo_ros1 LANGUAGES C CXX)
  set(FAST_LIVO_ROS1_ROOT_LAYOUT FALSE)
endif()

find_package(catkin REQUIRED COMPONENTS
  roscpp std_msgs sensor_msgs geometry_msgs nav_msgs visualization_msgs std_srvs
  tf2_ros cv_bridge pcl_conversions image_transport gnss_comm livox_ros_driver)
find_package(fast_livo_core CONFIG REQUIRED)

if(NOT FAST_LIVO_ROS1_ROOT_LAYOUT)
  catkin_package()
endif()
add_executable(fastlivo_mapping src/main.cpp src/outputs.cpp src/inputs.cpp src/config.cpp)
target_compile_features(fastlivo_mapping PRIVATE cxx_std_17)
target_include_directories(fastlivo_mapping PRIVATE include ${CMAKE_CURRENT_SOURCE_DIR} ${catkin_INCLUDE_DIRS})
target_link_libraries(fastlivo_mapping PRIVATE fast_livo::core ${catkin_LIBRARIES})
add_dependencies(fastlivo_mapping ${catkin_EXPORTED_TARGETS})
install(TARGETS fastlivo_mapping RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
if(NOT FAST_LIVO_ROS1_ROOT_LAYOUT)
  install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
endif()

if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(imu_conversion_test test/imu_conversion_test.cpp)
  if(TARGET imu_conversion_test)
    target_include_directories(imu_conversion_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${catkin_INCLUDE_DIRS})
    target_link_libraries(imu_conversion_test fast_livo::core ${catkin_LIBRARIES})
  endif()
  find_package(catkin REQUIRED COMPONENTS message_filters tf nav_msgs sensor_msgs)
  add_executable(backend_pairing_ros1_test test/backend_pairing_ros1_test.cpp)
  set_target_properties(backend_pairing_ros1_test PROPERTIES CXX_STANDARD 17 CXX_EXTENSIONS OFF)
  # 旧 tf 对照必须使用与默认核心一致的浮点编译模型。
  target_compile_options(backend_pairing_ros1_test PRIVATE -O3)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm|aarch64|ARM|AARCH64)")
    target_compile_options(backend_pairing_ros1_test PRIVATE -march=armv8-a -ffast-math)
  else()
    target_compile_options(backend_pairing_ros1_test PRIVATE -funroll-loops)
  endif()
  target_include_directories(backend_pairing_ros1_test PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(backend_pairing_ros1_test PRIVATE fast_livo::core gtsam ${catkin_LIBRARIES})
  add_test(NAME backend_pairing_ros1 COMMAND backend_pairing_ros1_test)
endif()
