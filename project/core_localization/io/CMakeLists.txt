cmake_minimum_required(VERSION 3.16)
project(fast_livo_io LANGUAGES C CXX)
find_package(fast_livo_core CONFIG REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(fast_livo_io src/dataset.cpp src/sample_codec.cpp)
target_compile_features(fast_livo_io PUBLIC cxx_std_17)
target_include_directories(fast_livo_io PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fast_livo_io PUBLIC fast_livo::core PRIVATE yaml-cpp)
install(TARGETS fast_livo_io EXPORT fast_livo_ioTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT fast_livo_ioTargets NAMESPACE fast_livo:: DESTINATION lib/cmake/fast_livo_io)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/fast_livo_ioConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/fast_livo_ioConfig.cmake INSTALL_DESTINATION lib/cmake/fast_livo_io)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fast_livo_ioConfig.cmake DESTINATION lib/cmake/fast_livo_io)
include(CTest)
if(BUILD_TESTING)
  add_executable(dataset_test test/dataset_test.cpp)
  target_link_libraries(dataset_test PRIVATE fast_livo_io)
  add_test(NAME dataset_roundtrip COMMAND dataset_test ${CMAKE_CURRENT_BINARY_DIR}/dataset-case)
endif()
