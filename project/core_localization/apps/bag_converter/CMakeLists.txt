cmake_minimum_required(VERSION 3.16)
project(fast_livo_bag_converter LANGUAGES C CXX)
find_package(fast_livo_io CONFIG REQUIRED)
find_package(catkin REQUIRED COMPONENTS rosbag cv_bridge sensor_msgs gnss_comm livox_ros_driver)
add_executable(fast_livo_bag_converter main.cpp conversion.cpp)
target_compile_features(fast_livo_bag_converter PRIVATE cxx_std_17)
target_include_directories(fast_livo_bag_converter PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(fast_livo_bag_converter PRIVATE fast_livo::fast_livo_io ${catkin_LIBRARIES})
install(TARGETS fast_livo_bag_converter RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(bag_converter_test ../../io/test/bag_converter_test.cpp conversion.cpp)
  target_include_directories(bag_converter_test PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(bag_converter_test PRIVATE fast_livo::fast_livo_io ${catkin_LIBRARIES})
  add_test(NAME bag_converter_roundtrip COMMAND bag_converter_test)
endif()
