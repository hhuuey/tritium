cmake_minimum_required(VERSION 3.16)
project(asset_protocol VERSION 1.0.0 LANGUAGES CXX)
find_package(nlohmann_json 3.11.2 CONFIG REQUIRED)
# Debian/Ubuntu Valijson 1.0 does not ship a ConfigVersion.cmake file.
# script/install.sh checks the package's minimum version.
find_package(valijson CONFIG REQUIRED)
add_library(asset_protocol STATIC src/json.cpp src/scanner.cpp src/protocol.cpp)
add_library(asset_protocol::asset_protocol ALIAS asset_protocol)
target_compile_features(asset_protocol PUBLIC cxx_std_17)
target_include_directories(asset_protocol PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_link_libraries(asset_protocol PUBLIC nlohmann_json::nlohmann_json PRIVATE valijson)
target_compile_options(asset_protocol PRIVATE -Wall -Wextra -Wpedantic)
option(ASSET_PROTOCOL_BUILD_TESTS "Build protocol conformance tests" OFF)
if(ASSET_PROTOCOL_BUILD_TESTS)
  enable_testing()
  add_executable(asset-protocol-test tests/protocol_test.cpp)
  target_compile_definitions(asset-protocol-test PRIVATE
    ASSET_PROTOCOL_FIXTURE_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../../fixtures/contract.json")
  target_link_libraries(asset-protocol-test PRIVATE asset_protocol::asset_protocol)
  add_test(NAME asset-protocol COMMAND asset-protocol-test)
  add_executable(asset-protocol-roundtrip tests/roundtrip.cpp)
  target_link_libraries(asset-protocol-roundtrip PRIVATE asset_protocol::asset_protocol)
endif()
install(TARGETS asset_protocol ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
